### V18__chain_tip_block_hash.sql

```sql
ALTER TABLE chain_tip ADD COLUMN block_hash TEXT;
ALTER TABLE chain_tip ALTER COLUMN block_height DROP NOT NULL;

WITH last_block AS (
    SELECT block_height, block_hash
    FROM locations
    ORDER BY block_height DESC
    LIMIT 1
)
UPDATE chain_tip SET
    block_height = (SELECT block_height FROM last_block),
    block_hash = (SELECT block_hash FROM last_block);
```

### V19__failed_blocks.sql

```sql
-- SKRYBITDEV-586: Persistent record of blocks the indexer skipped due to
-- parse/compress/standardize/download failures. The pipeline advances past
-- these blocks to avoid stalling; rows here track what needs manual retry
-- after a patch is deployed (via the `retry-failed` CLI command).

CREATE TABLE failed_blocks (
    block_height        NUMERIC PRIMARY KEY,
    block_hash          TEXT,
    error_kind          TEXT NOT NULL,          -- 'parse' | 'compress' | 'standardize' | 'download'
    error_message       TEXT NOT NULL,
    retry_count         INT NOT NULL DEFAULT 0,
    last_attempt_at     TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    resolved_at         TIMESTAMPTZ
);

CREATE INDEX failed_blocks_unresolved_idx
    ON failed_blocks (block_height)
    WHERE resolved_at IS NULL;
```
